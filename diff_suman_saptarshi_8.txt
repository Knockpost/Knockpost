+import 'package:flutter/services.dart';

+/// Centralized haptic feedback helper for Knockpost.
+/// Mirrors the haptic patterns used by Instagram, Facebook, X, and Substack.
+class KnockHaptics {
+  KnockHaptics._();
+
+  /// Light tap ΓÇö buttons, nav items, icon taps, send, toggles.
+  static void tap() => HapticFeedback.lightImpact();
+
+  /// Medium impact ΓÇö reactions, double-tap like, threshold events, long-press open.
+  static void impact() => HapticFeedback.mediumImpact();
+
+  /// Selection tick ΓÇö hovering over emoji/options during drag, toggling states.
+  static void selection() => HapticFeedback.selectionClick();
+
+  /// Heavy impact ΓÇö destructive actions (delete, remove).
+  static void heavy() => HapticFeedback.heavyImpact();
+
+  /// Full vibrate ΓÇö error states.
+  static void vibrate() => HapticFeedback.vibrate();
+}

+class _FastLongPressTileState extends State<_FastLongPressTile> {

+  void _cancelTimer() {
+    _timer?.cancel();
+    _timer = null;

+  @override
+  void dispose() {
+    _cancelTimer();
+    super.dispose();
+  }

+  @override
+  Widget build(BuildContext context) {
+    return GestureDetector(
+      behavior: HitTestBehavior.opaque,
+      onTapDown: (details) {

+      },
+      onTapUp: (_) => _cancelTimer(),
+      onTapCancel: () => _cancelTimer(),
+      onHorizontalDragEnd: (details) {
+        _cancelTimer();
+        if (details.primaryVelocity! > 100) {
+          widget.onSwipeReply();
+        }
+      },
+      child: widget.child,
